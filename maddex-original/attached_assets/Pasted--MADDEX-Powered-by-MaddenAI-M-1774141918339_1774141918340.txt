function getPlaceholderResponse(query) {
  const q = query.toLowerCase();
  if (q.includes("btc") || q.includes("bitcoin") || q.includes("crypto")) return PLACEHOLDER_RESPONSES.btc;
  if (q.includes("nvda") || q.includes("nvidia")) return PLACEHOLDER_RESPONSES.nvda;
  if (q.includes("super") || q.includes("superannuation")) return PLACEHOLDER_RESPONSES.super;
  return PLACEHOLDER_RESPONSES.default;
}

async function callMaddenAI(messages) {
  const apiKey = import.meta.env.VITE_ANTHROPIC_API_KEY;

  // No API key — return placeholder after realistic delay
  if (!apiKey) {
    await new Promise(r => setTimeout(r, 1200));
    const lastMsg = messages[messages.length - 1]?.content || "";
    return getPlaceholderResponse(lastMsg);
  }

  // Live Claude API call
  try {
    const response = await fetch("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "x-api-key": apiKey,
        "anthropic-version": "2023-06-01",
        "anthropic-dangerous-direct-browser-access": "true",
      },
      body: JSON.stringify({
        model: "claude-sonnet-4-20250514",
        max_tokens: 1024,
        system: SYSTEM_PROMPT,
        messages,
      }),
    });
    const data = await response.json();
    const text = data.content?.filter(b => b.type === "text").map(b => b.text).join("") || "";
    const clean = text.replace(/```json|```/g, "").trim();
    return JSON.parse(clean);
  } catch {
    return PLACEHOLDER_RESPONSES.default;
  }
}

// CHAT SCREEN
// ============================================================
function ChatScreen() {
  const [messages, setMessages] = useState([]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const [apiLive] = useState(!!import.meta.env.VITE_ANTHROPIC_API_KEY);
  const endRef = useRef(null);
  const inputRef = useRef(null);

  useEffect(() => {
    endRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages, loading]);

  const send = async (text) => {
    const q = text || input.trim();
    if (!q || loading) return;
    setInput("");

    const newMsgs = [...messages, { role: "user", content: q }];
    setMessages(newMsgs);
    setLoading(true);

    const apiMessages = newMsgs.map(m => ({ role: m.role, content: m.content }));
    const result = await callMaddenAI(apiMessages);

    setMessages([...newMsgs, { role: "assistant", data: result }]);
    setLoading(false);
    inputRef.current?.focus();
  };

  return (
    <div style={{ display: "flex", flexDirection: "column", height: "100%", overflow: "hidden" }}>
      {/* HEADER */}
      <div style={S.screenHeader}>
        <div>
          <div style={S.screenTitle}>MaddenAI</div>
          <div style={S.screenSubtitle}>Financial Intelligence Engine</div>
        </div>
        <div style={{ ...S.badge, background: apiLive ? C.posDim : C.goldDim, color: apiLive ? C.pos : C.gold }}>
          {apiLive ? "● Live" : "● Demo mode"}
        </div>
      </div>

      {/* MESSAGES */}
      <div style={{ flex: 1, overflowY: "auto", padding: "16px" }}>
        {messages.length === 0 && (
          <div style={{ textAlign: "center", padding: "32px 0" }}>
            <div style={{ fontFamily: "'Syne',sans-serif", fontSize: 22, fontWeight: 700, color: C.text, marginBottom: 8 }}>
              Ask MaddenAI anything.
            </div>
            <div style={{ fontSize: 13.5, color: C.textMuted, marginBottom: 24, lineHeight: 1.6 }}>
              Markets · Super · Investing · Tax · Crypto
            </div>
            <div style={{ display: "flex", flexWrap: "wrap", gap: 8, justifyContent: "center" }}>
              {QUICK_PROMPTS.map(p => (
                <button key={p} style={S.chip} onClick={() => send(p)}>{p}</button>
              ))}
            </div>
          </div>
        )}

        {messages.map((m, i) => (
          <div key={i}>
            {m.role === "user" ? (
              <div style={{ display: "flex", justifyContent: "flex-end", marginBottom: 12 }}>
                <div style={S.userBubble}>{m.content}</div>
              </div>
            ) : (
              <AIResponseCard data={m.data} />
            )}
          </div>
        ))}

        {loading && (
          <div style={{ display: "flex", gap: 8, alignItems: "center", padding: "12px 0" }}>
            <div style={S.aiDot} />
            <div style={{ display: "flex", gap: 4 }}>
              {[0, 1, 2].map(i => (
                <div key={i} style={{ width: 6, height: 6, borderRadius: "50%", background: C.accent, animation: "pulse 1.2s ease-in-out infinite", animationDelay: `${i * 200}ms` }} />
              ))}
            </div>
            <span style={{ fontSize: 12, color: C.textMuted }}>MaddenAI is analysing...</span>
          </div>
        )}
        <div ref={endRef} />
      </div>

      {/* INPUT */}
      <div style={{ padding: "12px 16px", borderTop: `1px solid ${C.cardBorder}` }}>
        {messages.length > 0 && (
          <div style={{ display: "flex", gap: 6, marginBottom: 10, overflowX: "auto", paddingBottom: 2 }}>
            {QUICK_PROMPTS.slice(0, 3).map(p => (
              <button key={p} style={{ ...S.chip, whiteSpace: "nowrap", flexShrink: 0 }} onClick={() => send(p)}>{p}</button>
            ))}
          </div>
        )}
        <div style={S.inputRow}>
          <textarea
            ref={inputRef}
            style={S.textarea}
            value={input}
            onChange={e => setInput(e.target.value)}
            onKeyDown={e => { if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); send(); } }}
            placeholder="Ask about markets, super, investing..."
            rows={1}
          />
          <button
            style={{ ...S.sendBtn, opacity: input.trim() && !loading ? 1 : 0.3 }}
            onClick={() => send()}
            disabled={!input.trim() || loading}
          >↑</button>
        </div>
        {!apiLive && (
          <div style={{ fontSize: 10.5, color: C.textMuted, textAlign: "center", marginTop: 8 }}>
            Demo mode — add VITE_ANTHROPIC_API_KEY to Replit Secrets to go live
          </div>
        )}
      </div>
    </div>
  );
}
